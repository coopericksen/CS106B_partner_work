A3. 

void duplicateNegatives(Queue<int>& q) {
    int startSize = q.size(); // store q size before modifying it
    for (int i = 0; i < startSize; i++) {
        int val = q.dequeue();
        q.enqueue(val);
        if (val < 0) {
            q.enqueue(val);   // double up on negative numbers
        }
	cout << q << endl;
    }
}
